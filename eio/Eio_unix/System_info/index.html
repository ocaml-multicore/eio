Module Eio_unix.System_info

Provides information about the system from e.g. uname(2).

  • since 1.7
type t = {
  1. sysname : string;
  2. release : string;
  3. version : string;
  4. machine : string;
}

Examples:

{ "sysname" = "Linux";        
  "release" = "7.2.8";
  "version" = "#1-NixOS SMP PREEMPT_DYNAMIC Fri Sep 25 14:37:14 UTC 2026";
  "machine" = "x86_64" }

{ "sysname" = "FreeBSD";
  "release" = "15.1-RELEASE";
  "version" = "FreeBSD 15.1-RELEASE releng/15.1-n283562-96841ea08dcf GENERIC";
  "machine" = "amd64" }

{ "sysname" = "Darwin";
  "release" = "24.5.0";
  "version" =
  "Darwin Kernel Version 24.5.0: Tue Apr 22 19:48:46 PDT 2025; root:xnu-11417.121.6~2/RELEASE_ARM64_T8103";
  "machine" = "arm64" }

{ "sysname" = "OpenBSD";
  "release" = "7.9";
  "version" = "GENERIC.MP#449";
  "machine" = "amd64" }

{ "sysname" = "Windows";
  "release" = "10.0.26100";
  "version" = "Windows Server 2025 Standard 24H2 (26100.4061)";
  "machine" = "x86_64" }
val get : unit -> t

Get information about the host system.

val dump : t Fmt.t
